CAS Mission State Engine --- class CASEngine: def __init__(self, callsign_jtac: str, callsign_aircraft: str): self.jtac = callsign_jtac self.aircraft = callsign_aircraft self.state = CASState.IDLE self.briefing: Optional[FiveLineBriefing] = None self.readback_confirmed = False def check_in(self, mnpopa_summary: str): """ขั้นตอนที่ 1: นักบินรายงานตัว (MNPOPA)""" self.state = CASState.CHECKED_IN return f"[{self.aircraft}] Check-in สำเร็จ: {mnpopa_summary}. สถานะ: {self.state.value}" def send_5line(self, briefing: FiveLineBriefing): """ขั้นตอนที่ 2: JTAC ส่งข้อมูล 5-Line Briefing""" if self.state != CASState.CHECKED_IN: raise Exception("ยังไม่ได้ Check-In ไม่สามารถส่ง 5-Line ได้") self.briefing = briefing self.state = CASState.BRIEFING_SENT return f"[{self.jtac}] ส่ง 5-Line Briefing สำเร็จ. รอการ Readback จากนักบิน..." def validate_readback(self, pilot_readback_fah: str, pilot_acknowledged_danger_close: bool) -> bool: """ขั้นตอนที่ 3: ตรวจสอบความถูกต้องของการ Readback""" if self.state != CASState.BRIEFING_SENT or not self.briefing: raise Exception("ไม่มี Briefing ให้ตรวจสอบ") # ตรวจสอบ Danger Close Acknowledgment if self.briefing.line_5.danger_close and not pilot_acknowledged_danger_close: print("❌ Validation Fail: นักบินไม่ได้ยืนยัน Danger Close!") return False self.readback_confirmed = True self.state = CASState.READBACK_VALIDATED print("✅ Validation Success: ข้อมูล Readback ถูกต้องครบถ้วน") return True def authorize_fire(self, command: FireCommand) -> str: """ขั้นตอนที่ 4: ออกคำสั่งลั่นไก (Fire Authorization)""" if self.state != CASState.READBACK_VALIDATED: return "❌ คำสั่งถูกปฏิเสธ: ต้องทวน Readback ให้ถูกต้องก่อนอนุมัติยิง" ctrl = self.briefing.line_1.control_type # ตรวจสอบ Rule การออกคำสั่งตาม Type of Control if ctrl in [ControlType.TYPE_1, ControlType.TYPE_2] and command != FireCommand.CLEARED_HOT: return f"❌ Rule Violation: {ctrl.value} ต้องใช้คำสั่ง 'Cleared Hot' เท่านั้น" elif ctrl == ControlType.TYPE_3 and command != FireCommand.CLEARED_TO_ENGAGE: return f"❌ Rule Violation: Type 3 ต้องใช้คำสั่ง 'Cleared to Engage' เท่านั้น" self.state = CASState.AUTHORIZED return f"[{self.jtac}] ออกคำสั่ง: '{command.value}'! สถานะ: {self.state.value}" #
